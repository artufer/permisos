async function guardarAusencia() { const trabajadorId = document.getElementById('aTrabajador').value; const tipo = document.getElementById('aTipo').value; const inicio = document.getElementById('aInicio').value; const fin = document.getElementById('aFin').value; const anioSel = document.getElementById('aAnio').value; const obs = document.getElementById('aObs').value.trim(); if (!trabajadorId || !inicio || !fin) { await showAlert('Datos incompletos', 'Complete trabajador, fecha inicio y fin.', 'warning'); return; } if (inicio > fin) { await showAlert('Fechas incorrectas', 'La fecha de inicio no puede ser posterior al fin.', 'warning'); return; } // ✅ VALIDACIÓN MEJORADA: Detectar solapamiento de fechas const permisosExistentes = ausencias.filter(a => a.trabajadorId === trabajadorId && a.inicio <= fin && a.fin >= inicio ); if (permisosExistentes.length > 0) { let mensaje = `⚠️ Ya existen permisos registrados que se solapan con estas fechas:\n\n`; permisosExistentes.forEach((p, idx) => { const tipoNombre = { vacaciones: '🏖️ Vacaciones', asuntos_propios: '📝 Asuntos propios', compensacion: '🔄 Compensación', familiar_primer: '👨‍‍👧 Familiar 1º', familiar_segundo: '👥 Familiar 2º', reposo: '🛏️ Reposo dom.', licencia: ' Licencia s/s', baja: ' Baja' }[p.tipo] || p.tipo; mensaje += `${idx + 1}. ${tipoNombre}: ${fmtDate(p.inicio)} - ${fmtDate(p.fin)} (${p.diasConsumidos} días)\n`; }); mensaje += `\n No se puede registrar este permiso. Elimine o modifique los permisos existentes primero.`; await showAlert('Fechas solapadas', mensaje, 'error'); return; // Bloquea completamente el registro } const fechaLimite = obtenerFechaLimite(anioSel, tipo); if ((tipo === 'vacaciones' || tipo === 'asuntos_propios') && fin > fechaLimite) { await showAlert('Fecha fuera de plazo', `Los permisos de ${tipo === 'vacaciones' ? 'vacaciones' : 'asuntos propios'} del año ${anioSel} se pueden disfrutar hasta el 31/01/${parseInt(anioSel)+1}.\n\nLa fecha fin (${fmtDate(fin)}) supera ese límite.`, 'warning'); return; } const anio = inicio.split('-')[0]; if (anio !== anioSel) { const ok = await showConfirm('Año diferente', `La fecha de inicio es del año ${anio} pero seleccionó el año ${anioSel}. ¿Continuar?`, 'warning'); if (!ok) return; } const cfg = configs[anio]; if (!cfg && tipo === 'compensacion') { const ok = await showConfirm('Sin configuración', `No hay configuración para ${anio}. ¿Continuar?`, 'warning'); if (!ok) return; } const calc = calcularDias(); if (!calc) return; const t = trabajadores.find(x => x.id === trabajadorId); const d = getDiasTrabajador(t, anio); if (tipo === 'vacaciones') { if (calc.vacaciones.diasNaturalesPeriodo < 7) { await showAlert('Periodo insuficiente', `El periodo total tiene ${calc.vacaciones.diasNaturalesPeriodo} días naturales. El mínimo es 7 días naturales (contando fines de semana adyacentes si aplica).`, 'warning'); return; } const periodosActuales = ausencias.filter(a => a.trabajadorId === trabajadorId && a.tipo === 'vacaciones' && a.inicio.startsWith(anio)); if (cfg && periodosActuales.length >= cfg.maxPeriodos) { const ok = await showConfirm('Máximo periodos', `Ya tiene ${periodosActuales.length} periodos (máx ${cfg.maxPeriodos}). ¿Continuar?`, 'warning'); if (!ok) return; } const vC = periodosActuales.reduce((s,a) => s + a.diasConsumidos, 0); if (vC + calc.dias > d.vacaciones) { const ok = await showConfirm('Supera vacaciones', `Se quedaría con ${vC + calc.dias}/${d.vacaciones} días. ¿Continuar?`, 'warning'); if (!ok) return; } } if (tipo === 'asuntos_propios') { const apC = ausencias.filter(a => a.trabajadorId === trabajadorId && a.tipo === 'asuntos_propios' && a.inicio.startsWith(anio)).reduce((s,a) => s + a.diasConsumidos, 0); if (apC + calc.dias > d.asuntosPropios) { const ok = await showConfirm('Supera AP', `Se quedaría con ${apC + calc.dias}/${d.asuntosPropios} días. ¿Continuar?`, 'warning'); if (!ok) return; } } if (tipo === 'compensacion') { const cC = ausencias.filter(a => a.trabajadorId === trabajadorId && a.tipo === 'compensacion' && a.inicio.startsWith(anio)).reduce((s,a) => s + a.diasConsumidos, 0); const cTotal = cfg ? cfg.compensacion : 0; if (cTotal > 0 && cC + calc.dias > cTotal) { const ok = await showConfirm('Supera compensación', `${cC + calc.dias}/${cTotal}. ¿Continuar?`, 'warning'); if (!ok) return; } } if (tipo === 'familiar_primer') { if (calc.dias > 5) { await showAlert('Supera máximo por solicitud', `Los permisos de familiar de 1er grado tienen un máximo de 5 días laborables por solicitud. Esta solicitud tiene ${calc.dias} días.`, 'warning'); return; } } if (tipo === 'familiar_segundo') { if (calc.dias > 4) { await showAlert('Supera máximo por solicitud', `Los permisos de familiar de 2º grado tienen un máximo de 4 días laborables por solicitud. Esta solicitud tiene ${calc.dias} días.`, 'warning'); return; } } if (tipo === 'reposo') { const rC = ausencias.filter(a => a.trabajadorId === trabajadorId && a.tipo === 'reposo' && a.inicio.startsWith(anio)).reduce((s,a) => s + a.diasConsumidos, 0); if (rC + calc.dias > 5) { await showAlert('Supera reposo domiciliario anual', `El reposo domiciliario tiene un máximo de 5 días laborables al año.\n\nYa tiene ${rC} días consumidos y esta solicitud son ${calc.dias} días (total: ${rC + calc.dias}).`, 'warning'); return; } } const nuevoId = 'A' + Date.now(); const data = { trabajadorId, tipo, inicio, fin, obs, diasConsumidos: calc.dias, diasNaturales: calc.naturales, anio }; ausencias.push({id: nuevoId, ...data}); await fsSet('ausencias', nuevoId, data); guardarLocal(); document.getElementById('aObs').value = ''; renderPermisosTrabajador(); mostrarDisponibles(); const nombres = {vacaciones:'Vacaciones', asuntos_propios:'Asuntos propios', compensacion:'Compensación', familiar_primer:'Familiar 1º', familiar_segundo:'Familiar 2º', reposo:'Reposo dom.', licencia:'Licencia s/s', baja:'Baja'}; toast('Permiso registrado', `${t.nombre} · ${nombres[tipo]} · ${calc.dias} días`, 'success'); }
